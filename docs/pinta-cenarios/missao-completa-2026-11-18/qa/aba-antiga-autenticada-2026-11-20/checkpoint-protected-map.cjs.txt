module.exports=async function(qa){
 const id='1cb291ea-b512-4e46-b3cd-e5d783454c3a';
 const result=await qa.modern.evaluate(async({namespace,id})=>{
  const name=namespace?'sistema-zero-pinta-'+namespace:'sistema-zero-pinta';
  const db=await new Promise((resolve,reject)=>{const r=indexedDB.open(name);r.onupgradeneeded=()=>{r.transaction.abort();reject(Error('Authority DB missing'))};r.onerror=()=>reject(Error('Authority DB unavailable'));r.onsuccess=()=>resolve(r.result)});
  try{
   const raw=await new Promise((resolve,reject)=>{const r=db.transaction('kv','readonly').objectStore('kv').get('pinta:local:document:'+id);r.onsuccess=()=>resolve(r.result);r.onerror=()=>reject(Error('Authority missing'))});

  }finally{db.close()}
 },{namespace:qa.ownedStorageNamespace,id});

 return {...result,readOnly:true,completeAuthoringDocumentRetained:false};
};
