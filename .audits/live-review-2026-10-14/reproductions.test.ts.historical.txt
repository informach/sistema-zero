const databaseUrl = await prepareTestDatabase('sistemazero_live_review')
const connection = databaseUrl ? createDbConnection(databaseUrl) : null
afterAll(async () => { await connection?.close() })

async function fixture(repository: LiveRepository = new InMemoryLiveRepository()) {
  const startsAt = new Date(Date.now() - 60_000).toISOString()
  const endsAt = new Date(Date.now() + 60_000).toISOString()
  let now = new Date(startsAt)
  const teacher = { id: randomUUID(), accountId: randomUUID(), name: 'Professora', staff: true }
  const child = { id: randomUUID(), accountId: randomUUID(), name: 'Criança', staff: false }
  const live = new LiveService(repository, () => now, { assertOwnership: async () => ({ name: child.name }) }, { mediaReady: true })
  const program = await live.createProgram(teacher, {
    title: 'Review', description: 'Reprodução isolada',
    lessons: [{ title: 'Encontro', instructions: 'Criar', starterProjectId: null, materials: [] }],
  })
  const { cohort, sessions: [session] } = await live.createCohort(teacher, program.id, {
    title: 'Review', timeZone: 'America/Sao_Paulo', capacity: 8, accessDays: 30,
    teacherIds: [teacher.id], schedule: [{ startsAt, endsAt }],
  })
  await live.updateCohort(teacher, cohort.id, 'open')
  await live.enroll(teacher, cohort.id, { accountId: child.accountId, learnerId: child.id, learnerName: child.name })
  await live.control(teacher, session!.id, 1, { kind: 'start' })
  return { repository, live, teacher, child, cohort, session: session!, setTime: (date: string) => { now = new Date(date) } }
}

describe('Evidence: lifecycle and replacement', () => {
  test('hard end denies media but leaves occurrence live', async () => {
    const f = await fixture()
    f.setTime(new Date(Date.parse(f.session.endsAt) + 1000).toISOString())
    const room = await f.live.room(f.teacher, f.session.id)
    expect(room.occurrence!.state).toBe('live')
    expect(room.access.canJoinMedia).toBe(false)
    await expect(f.live.authorizeMedia(f.teacher, f.session.occurrenceId, randomUUID())).rejects.toThrow()
  })

  test('replacement agenda still exposes the original date', async () => {
    const f = await fixture()
    await f.live.control(f.teacher, f.session.id, 2, { kind: 'end' })
    const startsAt = new Date(Date.now() + 86_400_000).toISOString()
    const replacement = await f.live.replaceOccurrence(f.teacher, f.session.id, {
      expectedVersion: 3, startsAt, endsAt: new Date(Date.now() + 90_000_000).toISOString(),
    })
    const catalog = await f.live.catalog(f.child)
    expect(catalog.sessions[0]!.occurrenceId).toBe(replacement.id)
    expect(catalog.sessions[0]!.startsAt).toBe(f.session.startsAt)
    expect(catalog.sessions[0]!.startsAt).not.toBe(replacement.startsAt)
  })

  test('withdrawal after replacement does not withdraw the old available replay', async () => {
    const f = await fixture()
    const reviewed = await f.live.setRecordingState(f.session.occurrenceId, {
      state: 'review', storageKey: `live/recordings/${f.session.occurrenceId}/review.mp4`,
    })
    await f.live.publishRecording(f.teacher, f.session.id, reviewed.version, true)
    await f.live.control(f.teacher, f.session.id, 2, { kind: 'end' })
    await f.live.replaceOccurrence(f.teacher, f.session.id, {
      expectedVersion: 3, startsAt: new Date(Date.now() + 86_400_000).toISOString(),
      endsAt: new Date(Date.now() + 90_000_000).toISOString(),
    })
    const current = await f.live.room(f.teacher, f.session.id)
    expect(current.occurrence!.recording.state).toBe('not_started')
    await f.live.publishRecording(f.teacher, f.session.id, current.occurrence!.recording.version, false)
    expect((await f.live.recording(f.child, f.session.id)).occurrenceId).toBe(f.session.occurrenceId)
  })
})

describe.skipIf(!connection)('Evidence: recorder reconciliation on local PostgreSQL', () => {
  test('an unanswered assistance request pauses recording and teacher cannot clear it', async () => {
    const f = await fixture(new DrizzleLiveRepository(connection!.db))
    const deviceId = randomUUID()
    await f.live.join(f.child, f.session.id, { deviceId, takeover: false })
    const room = await f.live.room(f.teacher, f.session.id)
    await f.live.control(f.teacher, f.session.id, room.occurrence!.version, { kind: 'admit', learnerId: f.child.id })
    const workspaceService = new LiveWorkspaceService(new DrizzleLiveWorkspaceRepository(connection!.db), f.live, () => new Date(f.session.startsAt))
    const workspace = await workspaceService.open(f.child, f.session.id, f.child.id)
    await workspaceService.claim(f.child, workspace.id, deviceId, false)
    await workspaceService.requestAssistance(f.teacher, workspace.id)
    const commands: any[] = []
    const server = Bun.serve({ port: 0, hostname: '127.0.0.1', async fetch(request) {
      const command = await request.json() as Record<string, unknown>
      commands.push(command)
      return Response.json({ ...command, state: 'paused', captureStopped: true })
    } })
    try {
      const provider = new OwnRecordingProvider(connection!.db, f.repository, f.live, {
        url: `http://127.0.0.1:${server.port}`, token: 'review-token-with-at-least-32-characters',
      })
      await provider.syncOccurrence(f.session.occurrenceId)
      expect(commands[0].desiredState).toBe('paused')
      const resumeFailure = await provider.command(f.teacher, f.session.id, 'resume').catch(error => error)
      expect(resumeFailure.code).toBe('RECORDING_ASSISTANCE_ACTIVE')
      const revokeFailure = await workspaceService.revoke(f.teacher, workspace.id, randomUUID()).catch(error => error)
      expect(revokeFailure).toBeInstanceOf(Error)
      expect((await workspaceService.get(f.teacher, workspace.id)).assistance?.status).toBe('requested')
    } finally {
      await connection!.db.update(liveRecordingJobs).set({ desiredState: 'stopped' }).where(eq(liveRecordingJobs.occurrenceId, f.session.occurrenceId))
      await server.stop(true)
    }
  })

  test('failed stop is excluded from the next reconciliation after recorder recovery', async () => {
    const f = await fixture(new DrizzleLiveRepository(connection!.db))
    let failing = true
    let requests = 0
    const server = Bun.serve({ port: 0, hostname: '127.0.0.1', async fetch(request) {
      requests++
      const command = await request.json() as Record<string, unknown>
      if (failing) return new Response('Transient failure', { status: 503 })
      return Response.json({ ...command, state: 'review', captureStopped: true })
    } })
    const provider = new OwnRecordingProvider(connection!.db, f.repository, f.live, {
      url: `http://127.0.0.1:${server.port}`, token: 'review-token-with-at-least-32-characters',
    })
    let stop: (() => Promise<void>) | undefined
    try {
      const failure = await provider.command(f.teacher, f.session.id, 'stop').catch(error => error)
      expect(failure).toBeInstanceOf(Error)
      expect(requests).toBe(1)
      failing = false
      stop = provider.start()
      await Bun.sleep(5600)
      expect(requests).toBe(1)
      const [job] = await connection!.db.select().from(liveRecordingJobs).where(eq(liveRecordingJobs.occurrenceId, f.session.occurrenceId))
      expect(job!.desiredState).toBe('stopped')
      expect(job!.confirmedState).not.toBe('review')
    } finally {
      await stop?.()
      await server.stop(true)
    }
  }, 40_000)
})
